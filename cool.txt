    private float[] rgb2hsl(int color) {
        float r = ((color >> 16) & 0xFF) / 255f;
        float g = ((color >> 8) & 0xFF) / 255f;
        float b = (color & 0xFF) / 255f;
        float mn = Math.min(Math.min(r, g), b);
        float mx = Math.max(Math.max(r, g), b);
        float d = mx - mn;
        float h = 0f, s = 0f;
        float l = (mx + mn) / 2f;
        if (d != 0f) {
            s = l < .5f ? d / (mx + mn) : d / (2f - mx - mn);
            float dr = ((mx - r) / 6f + d / 2f) / d;
            float dg = ((mx - g) / 6f + d / 2f) / d;
            float db = ((mx - b) / 6f + d / 2f) / d;
            if (r == mx) h = db - dg;
            else if (g == mx) h = (1f/3f) + dr - db;
            else h = (2f/3f) + dg - dr;
            if (h < 0f) h += 1f;
            if (h > 1f) h -= 1f;
        }
        return new float[]{h, s, l};
    }

    private int hsl2rgb(float h, float s, float l) {
        float r = l, g = l, b = l;
        float v = (l <= .5f) ? (l * (1f + s)) : (l + s - l * s);
        if (v > 0f) {
            float m = l + l - v;
            float sv = (v - m) / v;
            h *= 6f;
            int sect = (int)h;
            float fract = h - sect;
            float vsf = v * sv * fract;
            float mid1 = m + vsf;
            float mid2 = v - vsf;
            switch (sect) {
                case 0: r = v; g = mid1; b = m; break;
                case 1: r = mid2; g = v; b = m; break;
                case 2: r = m; g = v; b = mid1; break;
                case 3: r = m; g = mid2; b = v; break;
                case 4: r = mid1; g = m; b = v; break;
                case 5: r = v; g = m; b = mid2; break;
            }
        }
        return 0xFF000000
                | (((int)(r * 255f) & 0xFF) << 16)
                | (((int)(g * 255f) & 0xFF) << 8)
                | ((int)(b * 255f) & 0xFF);
    }

    private void startBytebeat(final int phase, final int sampleRate, final long durationMs) {
        stopBytebeat();
        bytebeatRunning = true;
        bytebeatThread = new Thread(new Runnable() {
            @Override public void run() {
                int minBuf = AudioTrack.getMinBufferSize(sampleRate,
                        AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_8BIT);
                if (minBuf <= 0) minBuf = sampleRate;
                AudioAttributes attrs = new AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_MEDIA)
                        .setContentType(AudioAttributes.CONTENT_TYPE_MUSIC)
                        .setFlags(AudioAttributes.FLAG_AUDIBILITY_ENFORCED)
                        .build();
                AudioFormat fmt = new AudioFormat.Builder()
                        .setEncoding(AudioFormat.ENCODING_PCM_8BIT)
                        .setSampleRate(sampleRate)
                        .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                        .build();
                AudioTrack track = new AudioTrack(
                        attrs, fmt, minBuf,
                        AudioTrack.MODE_STREAM,
                        AudioManager.AUDIO_SESSION_ID_GENERATE);
                bytebeatTrack = track;
                try { track.play(); } catch (Exception e) { bytebeatRunning = false; return; }
                int chunk = 2048;
                byte[] buf = new byte[chunk];
                long startTime = System.currentTimeMillis();
                long t = 0;
                while (bytebeatRunning && !finished) {
                    if (durationMs > 0 && (System.currentTimeMillis() - startTime) >= durationMs) {
                        bytebeatRunning = false;
                        break;
                    }
                    for (int i = 0; i < chunk; i++) {
                        int tt = (int)(t + i);
                        int v = 0;
                        switch (phase) {
                            case 1:
                                v = (tt & (tt >> 8));
                                break;
                            case 2:
                                v = (-tt >> 5 & ((tt >> 2) * (tt >> 5)));
                                break;
                            case 3:
                                v = (tt * (tt & (tt >> 12))) / 256;
                                break;
                            case 4:
                                v = (tt * (tt ^ tt + (tt >> 15 | 1) ^ (tt - 1280 ^ tt) >> 10));
                                break;
                            case 5:
                                v = (((tt >>> 5) | (tt >>> 4)) * tt) >> (tt >>> 16);
                                break;
                        }
                        buf[i] = (byte)(v & 0xFF);
                    }
                    try { track.write(buf, 0, chunk); } catch (Exception e) { break; }
                    t += chunk;
                }
                try { track.stop(); track.release(); } catch (Exception e) {}
                bytebeatTrack = null;
            }
        });
        bytebeatThread.start();
    }

    private void stopBytebeat() {
        bytebeatRunning = false;
        if (bytebeatThread != null) {
            try { bytebeatThread.join(200); } catch (InterruptedException e) {}
            bytebeatThread = null;
        }
        if (bytebeatTrack != null) {
            try { bytebeatTrack.stop(); bytebeatTrack.release(); } catch (Exception e) {}
            bytebeatTrack = null;
        }
    }

    private void drawXorPattern() {
        if (xorBuf == null) return;
        int[] pixels = new int[xorW * xorH];
        xorT += 2;
        int halfW = xorW / 2;
        int halfH = xorH / 2;
        for (int y = 0; y < xorH; y++) {
            int cy = y - halfH;
            int ay = Math.abs(cy);
            for (int x = 0; x < xorW; x++) {
                int cx = x - halfW;
                int ax = Math.abs(cx);
                int xorVal = (x ^ y) + xorT;
                int sierpMask = ((x & y) == 0) ? 1 : 0;
                int sierpXor = (ax ^ ay) + xorT;
                int finalVal = (sierpMask == 1) ? xorVal : sierpXor;
                float hue = (finalVal * 2.0f) % 360f;
                if (hue < 0f) hue += 360f;
                int hi = (int) (hue / 60f) % 6;
                float f = hue / 60f - (float) Math.floor(hue / 60f);
                float p = 0f, q = 1f - f, t = f;
                float rf, gf, bf;
                switch (hi) {
                    case 0: rf = 1f; gf = t; bf = p; break;
                    case 1: rf = q; gf = 1f; bf = p; break;
                    case 2: rf = p; gf = 1f; bf = t; break;
                    case 3: rf = p; gf = q; bf = 1f; break;
                    case 4: rf = t; gf = p; bf = 1f; break;
                    default: rf = 1f; gf = p; bf = q; break;
                }
                int outR = (int) (rf * 255f) & 0xFF;
                int outG = (int) (gf * 255f) & 0xFF;
                int outB = (int) (bf * 255f) & 0xFF;
                pixels[y * xorW + x] = (255 << 24) | (outR << 16) | (outG << 8) | outB;
            }
        }
        xorBuf.setPixels(pixels, 0, xorW, 0, 0, xorW, xorH);
        Rect dst = new Rect(0, 0, screenWidth, screenHeight);
        canvas.drawBitmap(xorBuf, null, dst, null);
    }

    private void drawIconLine(Bitmap icon, int x0, int y0, int x1, int y1) {
        int dx = Math.abs(x1 - x0);
        int dy = -Math.abs(y1 - y0);
        int sx = (x0 < x1) ? 1 : -1;
        int sy = (y0 < y1) ? 1 : -1;
        int error = dx + dy;
        int i = 0;

    private int hsvToRgb(float hue, float sat, float val) {
        if (hue < 0) hue += 360;
        if (hue >= 360) hue -= 360;
        int hi = (int) (hue / 60f) % 6;
        float f = hue / 60f - (float) Math.floor(hue / 60f);
        float p = val * (1f - sat);
        float q = val * (1f - f * sat);
        float t = val * (1f - (1f - f) * sat);
        float rf, gf, bf;
        switch (hi) {
            case 0: rf = val; gf = t; bf = p; break;
            case 1: rf = q; gf = val; bf = p; break;
            case 2: rf = p; gf = val; bf = t; break;
            case 3: rf = p; gf = q; bf = val; break;
            case 4: rf = t; gf = p; bf = val; break;
            default: rf = val; gf = p; bf = q; break;
        }
        return ((int) (rf * 255f) & 0xFF) << 16
                | ((int) (gf * 255f) & 0xFF) << 8
                | ((int) (bf * 255f) & 0xFF);
    }
